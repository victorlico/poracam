# config.txt — Poracam v0.5
# Formato: chave=valor
# Linhas começando com # são comentários.

# Modo de execução.
# Na v0.5, apenas "single" é suportado.
# single = executa uma gravação e encerra.
run_mode=single

# Duração de cada gravação, em segundos.
# Exemplos:
# 60  = 1 minuto
# 300 = 5 minutos
# 600 = 10 minutos
record_duration_s=60

# Período total do ciclo, em segundos.
# Ainda não executa loop na v0.5, mas já é validado.
# Exemplo: 300 = configuração compatível com "gravar 1 min a cada 5 min".
# Regra: cycle_period_s precisa ser maior ou igual a record_duration_s.
cycle_period_s=300

# Diretório onde os arquivos serão salvos.
# IMPORTANTE:
# - Se este config.txt estiver no armazenamento externo em PORACAM/config.txt
#   e media_dir ficar comentado/ausente, o Poracam salvará automaticamente em PORACAM/media.
# - Se quiser forçar um caminho específico, descomente e ajuste:
# media_dir=/home/fishcam/poracam/media

# Limite máximo de uso do armazenamento antes de gravar.
# Se o filesystem estiver com uso maior ou igual a este valor, a gravação não inicia.
max_storage_percent=95

# Configuração de armazenamento externo.
prefer_external_storage=true
allow_local_fallback=true

# Configuração de vídeo.
width=1280
height=720
fps=30
bitrate=2500000

# Configuração de áudio.
# Use "default" se já estiver funcionando.
# Para descobrir dispositivos:
# arecord -l
# Exemplo alternativo:
# audio_device=plughw:1,0
audio_device=default
audio_format=cd

# Se false, o .h264 temporário será apagado depois que o .mp4 for gerado.
# Padrão recomendado: false
keep_h264=false

# Atraso intencional antes de iniciar o vídeo, após iniciar o áudio.
# 0.2 significa que o áudio começa cerca de 0,2 s antes do vídeo.
# Use 0 para iniciar ambos quase em sequência.
start_delay_s=0.2

# Tempo extra para aguardar raspivid/arecord finalizarem.
extra_timeout_s=15
